🤔 什么是大宽表？

大宽表（Wide Table / Flattened Table） 是将星型模型中的事实表和所有常用维度表预先 JOIN 成一张"扁平化"大表。每一行包含分析所需的全部信息，查询时不需要任何 JOIN。

直观对比

星型模型查询（7 表 JOIN）：

SELECT p.brand, SUM(od.sub_total)
FROM fact_order_details od
JOIN fact_orders o         ON ...
JOIN dim_product p         ON ...
JOIN dim_product_category pc ON ...
JOIN dim_customer c        ON ...
JOIN dim_region r          ON ...
JOIN dim_channel ch        ON ...    -- 😰 7 次 JOIN！
WHERE pc.category_name = '手机通讯'
  AND r.region_name = '华东'
  AND c.customer_level = '金卡'
  AND ch.channel_name = 'App 商城'
GROUP BY p.brand;

大宽表查询（0 JOIN）：

xxxxxxxxxx
SELECT brand, SUM(sub_total)          -- 🎉 零 JOIN！
FROM wide_order_details
WHERE category_name = '手机通讯'
  AND region_name = '华东'
  AND customer_level = '金卡'
  AND channel_name = 'App 商城'
GROUP BY brand;

📐 DDL：wide_order_details（商品粒度 — 最常用）

​x
-- ============================================================
-- 商品粒度大宽表（核心分析表）
-- 粒度：一行 = 一个订单中的一个商品行
-- 用途：品类分析、品牌分析、商品销售趋势、毛利分析
-- 数据量：约 149,000 行
-- ============================================================
CREATE TABLE wide_order_details AS
SELECT
    -- ============ 主键与业务键 ============
    od.order_detail_key,
    od.order_detail_id,
    od.order_key,
    o.order_id,
​
    -- ============ 日期维度（退化自 dim_date）============
    dd.date_value        AS order_date,
    dd.year              AS order_year,
    dd.quarter           AS order_quarter,
    dd.month             AS order_month,
    dd.month_name        AS order_month_name,
    dd.day_of_week       AS order_weekday,
    dd.day_of_week_name  AS order_weekday_name,
    dd.is_weekend        AS is_weekend,
    dd.is_holiday        AS is_holiday,
​
    -- ============ 客户维度（退化自 dim_customer）============
    c.customer_id,
    c.customer_level,     -- '普通' / '银卡' / '金卡' / '钻石'
    c.age_group,          -- '18-25' / '26-35' / '36-45' / '46+'
    c.gender,             -- 0=未知 1=男 2=女
​
    -- ============ 商品维度（退化自 dim_product）============
    p.product_id,
    p.product_name,
    p.brand,
    p.unit,
​
    -- ============ 品类维度（退化自 dim_product_category）============
    pc.category_name,
    pc.category_level,    -- 1=一级品类 2=二级品类 3=三级品类
​
    -- ============ 区域维度（退化自 dim_region）============
    r.region_name,        -- '华东' / '华北' / '华南' / '西南' / '华中' / '东北' / '西北'
    r.region_level,       -- 1=全国 2=大区 3=省市 4=区县
    r.region_path,        -- '中国/华东/浙江省/杭州市'
​
    -- ============ 渠道维度（退化自 dim_channel）============
    ch.channel_name,      -- 'App 商城' / '微信小程序' / 'PC 官网' / '线下门店'
    ch.channel_type,      -- '线上' / '线下' / '分销'
​
    -- ============ 订单状态维度（退化自 dim_order_status）============
    os.status_name        AS order_status,   -- '已完成' / '已退款' 等
    os.status_type        AS status_type,    -- '已完成' / '已退款' 等
​
    -- ============ 支付方式维度（退化自 dim_payment_method）============
    pm.method_name        AS payment_method, -- '微信支付' / '支付宝' 等
​
    -- ============ 明细级度量值 ============
    od.quantity,          -- 购买数量
    od.unit_price,        -- 成交单价（下单时实际价格）
    od.original_unit_price, -- 商品原价（折扣前）
    od.discount,          -- 行级折扣金额
    od.sub_total,         -- 行小计 = unit_price * quantity - discount
    od.cost_price,        -- 成本价（快照，用于毛利计算）
​
    -- ============ 订单级度量值（冗余，避免回头 JOIN）============
    o.actual_amount       AS order_actual_amount, -- 订单实付金额
    o.shipping_fee,       -- 运费
    o.is_first_order,     -- 是否首单 0/1
​
    -- ============ 预计算字段 ============
    ROUND(od.sub_total - od.cost_price * od.quantity, 2) AS gross_profit,
    ROUND(od.discount / NULLIF(od.original_unit_price * od.quantity, 0) * 100, 2) AS discount_rate
​
FROM fact_order_details od
JOIN fact_orders o          ON od.order_key = o.order_key
JOIN dim_date dd            ON od.order_date_key = dd.date_key
JOIN dim_product p          ON od.product_key = p.product_key
JOIN dim_product_category pc ON p.category_key = pc.category_key
JOIN dim_customer c         ON o.customer_key = c.customer_key
JOIN dim_region r           ON o.region_key = r.region_key
JOIN dim_channel ch         ON o.channel_key = ch.channel_key
JOIN dim_order_status os    ON o.order_status_key = os.order_status_key
LEFT JOIN dim_payment_method pm ON o.payment_method_key = pm.payment_method_key;

📊 执行时间：约 3-5 秒（数据量 15 万行），结果：~149,000 行 × 35 列

📐 DDL：wide_orders（订单粒度）

xxxxxxxxxx
-- ============================================================
-- 订单粒度大宽表
-- 粒度：一行 = 一个订单
-- 用途：GMV 分析、客单价分析、渠道分析、区域分析
-- 数据量：约 50,000 行
-- ============================================================
CREATE TABLE wide_orders AS
SELECT
    o.order_key,
    o.order_id,
​
    -- 日期维度
    dd.date_value        AS order_date,
    dd.year              AS order_year,
    dd.quarter           AS order_quarter,
    dd.month             AS order_month,
    dd.month_name        AS order_month_name,
    dd.is_weekend        AS is_weekend,
​
    -- 客户维度
    c.customer_id,
    c.customer_level,
    c.age_group,
    c.gender,
    c.total_order_amount AS customer_lifetime_value,
    c.total_order_count  AS customer_lifetime_orders,
​
    -- 区域维度
    r.region_name,
    r.region_level,
    r.region_path,
​
    -- 渠道维度
    ch.channel_name,
    ch.channel_type,
​
    -- 订单状态
    os.status_name       AS order_status,
    os.status_type       AS status_type,
​
    -- 支付方式
    pm.method_name       AS payment_method,
​
    -- 订单度量值
    o.item_count,
    o.item_sku_count,
    o.original_amount,
    o.discount_amount,
    o.coupon_amount,
    o.shipping_fee,
    o.actual_amount,
    o.paid_amount,
    o.refund_amount,
    o.is_first_order,
​
    -- 预计算字段
    ROUND(o.actual_amount / NULLIF(o.item_sku_count, 0), 2) AS avg_sku_price,
    ROUND(o.discount_amount / NULLIF(o.original_amount, 0) * 100, 2) AS discount_rate
​
FROM fact_orders o
JOIN dim_date dd          ON o.order_date_key = dd.date_key
JOIN dim_customer c       ON o.customer_key = c.customer_key
JOIN dim_region r         ON o.region_key = r.region_key
JOIN dim_channel ch       ON o.channel_key = ch.channel_key
JOIN dim_order_status os  ON o.order_status_key = os.order_status_key
LEFT JOIN dim_payment_method pm ON o.payment_method_key = pm.payment_method_key;

🚀 宽表查询实战：重做第 4 章 10 题

Q1：2026 年 5 月总销售额

xxxxxxxxxx
SELECT ROUND(SUM(sub_total), 2) AS 总销售额
FROM wide_order_details
WHERE order_year = 2026 AND order_month = 5
  AND order_status != '已退款';

Q3：上月销售额最高的 5 个品类

xxxxxxxxxx
SELECT category_name AS 品类,
       ROUND(SUM(sub_total), 2) AS 销售额
FROM wide_order_details
WHERE order_year = 2026 AND order_month = 5
  AND order_status != '已退款'
GROUP BY category_name
ORDER BY 销售额 DESC
LIMIT 5;

Q8：手机品类下各品牌的销售额排名

xxxxxxxxxx
SELECT brand AS 品牌,
       ROUND(SUM(sub_total), 2) AS 销售额,
       SUM(quantity) AS 销量,
       COUNT(DISTINCT order_key) AS 订单数
FROM wide_order_details
WHERE category_name = '手机通讯'
  AND order_status != '已退款'
GROUP BY brand
ORDER BY 销售额 DESC;

Q9：华东区金卡会员通过 App 购买手机品牌 TOP5

xxxxxxxxxx
-- 🎉 0 次 JOIN！原来需要 7 表 JOIN 的查询现在只需单表 WHERE！
SELECT brand AS 品牌,
       ROUND(SUM(sub_total), 2) AS 销售额
FROM wide_order_details
WHERE category_name = '手机通讯'
  AND region_name = '华东'
  AND customer_level = '金卡'
  AND channel_name = 'App 商城'
  AND order_status != '已退款'
GROUP BY brand
ORDER BY 销售额 DESC
LIMIT 5;

Q10：2026 年各品类月度退款率趋势

xxxxxxxxxx
SELECT category_name AS 品类,
       order_month AS 月份,
       COUNT(DISTINCT order_key) AS 总订单数,
       ROUND(SUM(CASE WHEN status_type = '已退款' THEN sub_total ELSE 0 END)
             / NULLIF(SUM(sub_total), 0) * 100, 2) AS 退款率
FROM wide_order_details
WHERE order_year = 2026
GROUP BY category_name, order_month
ORDER BY category_name, order_month;

    table_name VARCHAR(100) PRIMARY KEY,
    last_refresh_time DATETIME
);
​
-- 插入增量数据（仅变更的数据）
INSERT INTO wide_order_details (...)

FROM fact_order_details od
JOIN ... (所有维度表)
WHERE od.updated_at > (
    SELECT last_refresh_time FROM wide_refresh_log
    WHERE table_name = 'wide_order_details'
);
​
-- 更新刷新日志
UPDATE wide_refresh_log SET last_refresh_time = NOW()
WHERE table_name = 'wide_order_details';

是否需要大宽表？
├─ 查询涉及 3 张以上表的 JOIN？ ─── 是 ──▶ 强烈建议宽表
